    36→interface StockEntry {
    37→  idProducto: number
    38→  idUbicacion: number

    44→interface InventoryItem {
    45→  id: number
    46→  sku: string
    47→  nombre: string
    48→  descripcion: string | null
    49→  idCategoria: number | null
    50→  idMarca: number | null

    62→function StatusBadge({ status }: { status: string }) {
    63→  if (status === 'out')
    64→    return (
    65→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-danger">
    66→        <XCircle className="h-3 w-3" /> Sin Stock
    67→      </span>
    68→    )
    69→  if (status === 'low')
    70→    return (
    71→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-warning">
    72→        <AlertTriangle className="h-3 w-3" /> Bajo Stock
    73→      </span>
    74→    )
    75→  return (
    76→    <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-success">
    77→      <Check className="h-3 w-3" /> OK
    78→    </span>
    79→  )
    80→}

    88→  // Batch selection state

    90→  const [batchDialogOpen, setBatchDialogOpen] = useState(false)
    91→  const [batchQty, setBatchQty] = useState('')
    92→  const [batchType, setBatchType] = useState<'add' | 'subtract'>('add')
    93→  const [batchLoading, setBatchLoading] = useState(false)
    94→
    95→  // Ajuste state

    97→  const [ajusteStock, setAjusteStock] = useState('')

   100→
   101→  // Traslado state

   104→  const [trasladoDestino, setTrasladoDestino] = useState('')

   106→  const [trasladoLoading, setTrasladoLoading] = useState(false)

   112→  const { data: products = [], isLoading } = useQuery({
   113→    queryKey: ['products-inventory'],

   117→  const { data: stockEntries = [] } = useQuery<StockEntry[]>({
   118→    queryKey: ['all-stock-inv'],

   122→  function toggleSelectAll() {
   123→    if (selectedIds.size === sortedInventory.length) {
   124→      setSelectedIds(new Set())
   125→    } else {
   126→      setSelectedIds(new Set(sortedInventory.map((p) => p.id)))
   127→    }
   128→  }
   129→
   130→  function toggleSelect(id: number) {
   131→    const next = new Set(selectedIds)
   132→    if (next.has(id)) {
   133→      next.delete(id)
   134→    } else {
   135→      next.add(id)
   136→    }
   137→    setSelectedIds(next)
   138→  }
   139→
   140→  function clearSelection() {
   141→    setSelectedIds(new Set())
   142→  }

   144→  async function submitBatchAdjust() {
   145→    if (selectedIds.size === 0 || !batchQty) return
   146→    const qty = parseInt(batchQty, 10)
   147→    if (isNaN(qty) || qty <= 0) {
   148→      toast.error('Cantidad inválida')
   149→      return
   150→    }
   151→
   152→    // Get stock entries for selected products
   153→    const selectedItems = sortedInventory
   154→      .filter((p) => selectedIds.has(p.id))
   155→      .flatMap((p) =>
   156→        p.stocks.map((s) => ({ idProducto: p.id, idUbicacion: s.idUbicacion }))
   157→      )
   158→
   159→    if (selectedItems.length === 0) {
   160→      toast.error('No hay stock entries para los productos seleccionados')
   161→      return
   162→    }
   163→
   164→    setBatchLoading(true)
   165→    try {
   166→      const res = await fetch('/api/wms/stock/batch-adjust', {
   167→        method: 'POST',
   168→        headers: { 'Content-Type': 'application/json' },
   169→        body: JSON.stringify({ items: selectedItems, cantidad: qty, tipo: batchType }),
   170→      })
   171→      if (!res.ok) {
   172→        const err = await res.json()
   173→        throw new Error(err.error || 'Error al ajustar stock')
   174→      }
   175→      toast.success(`Stock ajustado para ${selectedIds.size} productos`)
   176→      setBatchDialogOpen(false)
   177→      setBatchQty('')
   178→      setSelectedIds(new Set())
   179→      invalidateInventory()
   180→    } catch (e: any) {
   181→      toast.error(e.message || 'Error al ajustar stock')
   182→    } finally {
   183→      setBatchLoading(false)
   184→    }
   185→  }

   189→    if (selected.length === 0) return
   190→    exportToCSV(
   191→      selected.map((p) => ({
   192→        producto: p.nombre,
   193→        sku: p.sku,
   194→        totalStock: p.totalStock,
   195→        valorTotal: p.valorTotal,
   196→      })),
   197→      'inventario-seleccion',
   198→      [
   199→        { key: 'producto', label: 'Producto' },
   200→        { key: 'sku', label: 'SKU' },
   201→        { key: 'totalStock', label: 'Stock Total' },
   202→        { key: 'valorTotal', label: 'Valor Total', format: 'currency' },
   203→      ],
   204→    )
   205→  }
   206→
   207→  const { data: allLocations = [] } = useQuery({
   208→    queryKey: ['all-locations'],
   209→    queryFn: () => fetch('/api/wms/ubicaciones?activos=true').then((r) => r.json()),
   210→  })
   211→
   212→  const { data: productMovements = [] } = useQuery({
   213→    queryKey: ['product-movements-sheet', sheetProduct?.id],

   218→  // Build inventory view
   219→  const inventory: InventoryItem[] = products.map((p: any) => {

   222→    const valorTotal = totalStock * p.costoUnitario

   224→    if (totalStock === 0) status = 'out'
   225→    else if (totalStock < p.stockMinimo) status = 'low'
   226→    return { ...p, totalStock, valorTotal, status, stocks }
   227→  })

   229→  const { data: inventoryStats } = useQuery({

   232→      totalProducts: d.totalProductos,
   233→      stockValue: d.valorTotalStock,
   234→      bajoStock: d.productosBajoStock?.count ?? 0,
   235→    })),
   236→  })
   237→
   238→  const totalValor = inventory.reduce((sum, p) => sum + p.valorTotal, 0)
   239→
   240→  function invalidateInventory() {
   241→    queryClient.invalidateQueries({ queryKey: ['products-inventory'] })
   242→    queryClient.invalidateQueries({ queryKey: ['all-stock-inv'] })
   243→    queryClient.invalidateQueries({ queryKey: ['inventory-stats'] })
   244→    queryClient.invalidateQueries({ queryKey: ['product-movements-sheet'] })
   245→  }

   247→  // --- Sorting ---
   248→  function handleSort(field: string) {
   249→    if (sortField === field) {
   250→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   251→    } else {
   252→      setSortField(field)
   253→      setSortDir('asc')
   254→    }
   255→  }
   256→  function getSortValue(item: InventoryItem, field: string): any {
   257→    if (field === 'margen') {
   258→      return item.precioVenta > 0 ? ((item.precioVenta - item.costoUnitario) / item.precioVenta * 100) : 0
   259→    }
   260→    return item[field as keyof InventoryItem]
   261→  }
   262→
   263→  const sortedInventory = [...inventory].sort((a, b) => {
   264→    let aVal: any = getSortValue(a, sortField)
   265→    let bVal: any = getSortValue(b, sortField)
   266→    if (typeof aVal === 'string') { aVal = aVal.toLowerCase(); bVal = (bVal as string).toLowerCase() }
   267→    if (aVal < bVal) return sortDir === 'asc' ? -1 : 1
   268→    if (aVal > bVal) return sortDir === 'asc' ? 1 : -1
   269→    return 0
   270→  })

   272→  // --- AJUSTE handlers ---
   273→  function openAjuste(item: InventoryItem) {
   274→    setAjusteProduct(item)
   275→    setAjusteStock(String(item.totalStock))
   276→    setAjusteMotivo('')

   279→  async function submitAjuste() {
   280→    if (!ajusteProduct || ajusteStock === '') return
   281→    const newStock = parseInt(ajusteStock, 10)
   282→    if (isNaN(newStock) || newStock < 0) {
   283→      toast.error('Cantidad inválida')
   284→      return
   285→    }
   286→
   287→    const locationId = ajusteProduct.stocks.length > 0
   288→      ? ajusteProduct.stocks[0].idUbicacion
   289→      : allLocations[0]?.id
   290→
   291→    if (!locationId) {
   292→      toast.error('No hay ubicaciones disponibles')
   293→      return
   294→    }
   295→
   296→    setAjusteLoading(true)
   297→    try {
   298→      const res = await fetch('/api/wms/movimientos', {
   299→        method: 'POST',
   300→        headers: { 'Content-Type': 'application/json' },
   301→        body: JSON.stringify({
   302→          idProducto: ajusteProduct.id,
   303→          idUbicacion: locationId,
   304→          idTipo: 3,
   305→          cantidad: newStock,
   306→          referencia: 'Ajuste manual',
   307→          observacion: ajusteMotivo || null,
   308→        }),
   309→      })
   310→      if (!res.ok) {
   311→        const err = await res.json()
   312→        throw new Error(err.error || 'Error al ajustar stock')
   313→      }
   314→      toast.success('Stock ajustado correctamente')
   315→      setAjusteProduct(null)
   316→      invalidateInventory()
   317→    } catch (e: any) {
   318→      toast.error(e.message || 'Error al ajustar stock')
   319→    } finally {
   320→      setAjusteLoading(false)
   321→    }
   322→  }
   323→
   324→  // --- TRASLADO handlers ---
   325→  function openTraslado(item: InventoryItem) {
   326→    if (item.stocks.length < 1) return
   327→    setTrasladoProduct(item)
   328→    setTrasladoOrigen(String(item.stocks[0].idUbicacion))
   329→    setTrasladoDestino('')
   330→    setTrasladoCantidad('')
   331→  }
   332→
   333→  function getOrigenStock(): number {
   334→    if (!trasladoProduct) return 0
   335→    const entry = trasladoProduct.stocks.find((s) => String(s.idUbicacion) === trasladoOrigen)
   336→    return entry?.cantidad ?? 0
   337→  }
   338→
   339→  async function submitTraslado() {
   340→    if (!trasladoProduct || !trasladoOrigen || !trasladoDestino || !trasladoCantidad) return
   341→    if (trasladoOrigen === trasladoDestino) {
   342→      toast.error('El origen y destino no pueden ser iguales')
   343→      return
   344→    }
   345→    const qty = parseInt(trasladoCantidad, 10)
   346→    if (isNaN(qty) || qty <= 0) {
   347→      toast.error('Cantidad inválida')
   348→      return
   349→    }
   350→    const origenStock = getOrigenStock()
   351→    if (qty > origenStock) {
   352→      toast.error(`Stock insuficiente en origen (disponible: ${origenStock})`)
   353→      return
   354→    }
   355→
   356→    setTrasladoLoading(true)
   357→    try {
   358→      const res = await fetch('/api/wms/stock/transfer', {
   359→        method: 'POST',
   360→        headers: { 'Content-Type': 'application/json' },
   361→        body: JSON.stringify({
   362→          idProducto: trasladoProduct.id,
   363→          idUbicacionOrigen: parseInt(trasladoOrigen, 10),
   364→          idUbicacionDestino: parseInt(trasladoDestino, 10),
   365→          cantidad: qty,
   366→        }),
   367→      })
   368→      if (!res.ok) {
   369→        const err = await res.json()
   370→        throw new Error(err.error || 'Error en traslado')
   371→      }
   372→      toast.success('Traslado realizado correctamente')
   373→      setTrasladoProduct(null)
   374→      invalidateInventory()
   375→    } catch (e: any) {
   376→      toast.error(e.message || 'Error en traslado')
   377→    } finally {
   378→      setTrasladoLoading(false)
   379→    }
   380→  }

   481→                  <SortableHeader field="margen" align="right" sortField={sortField} sortDir={sortDir} onSort={handleSort} className="hidden lg:table-cell">Margen %</SortableHeader>
   482→                  <TableHead className="text-xs text-center">Estado</TableHead>
   483→                  <TableHead className="text-xs text-right">Acciones</TableHead>
   484→                </TableRow>
   485→              </TableHeader>
   486→              <TableBody>
   487→                {isLoading && Array.from({ length: 8 }).map((_, i) => (

   489→                ))}
   490→                {!isLoading && sortedInventory.length === 0 && (
   491→                  <TableRow><TableCell colSpan={8} className="text-center text-muted-foreground py-8">Sin productos en inventario</TableCell></TableRow>
   492→                )}
   493→                {!isLoading && sortedInventory.map((p, idx) => {
   494→                  const margen = p.precioVenta > 0 ? ((p.precioVenta - p.costoUnitario) / p.precioVenta * 100) : 0
   495→                  return (
   496→                    <TableRow key={p.id} className={cn('hover:bg-muted/50 cursor-pointer', idx % 2 === 1 && 'bg-muted/20', selectedIds.has(p.id) && 'bg-primary/5')} onClick={() => openSheet(p)}>
   497→                      <TableCell className="text-xs py-2" onClick={(e) => e.stopPropagation()}>
   498→                        <Checkbox checked={selectedIds.has(p.id)} onCheckedChange={() => toggleSelect(p.id)} />
   499→                      </TableCell>
   500→                      <TableCell className="text-xs font-medium py-2 max-w-[200px] truncate">{p.nombre}</TableCell>
   501→                      <TableCell className="text-xs font-mono py-2">{p.sku}</TableCell>
   502→                      <TableCell className="text-xs text-center font-mono py-2">{p.totalStock}</TableCell>
   503→                      <TableCell className="text-xs text-right py-2 hidden md:table-cell">{formatCurrency(p.valorTotal)}</TableCell>
   504→                      <TableCell className="text-xs text-right py-2 hidden lg:table-cell">
   505→                        <span className={cn(
   506→                          'font-mono',
   507→                          margen >= 30 ? 'text-emerald-600 dark:text-emerald-400' :
   508→                          margen >= 15 ? 'text-amber-600 dark:text-amber-400' :
   509→                          'text-red-600 dark:text-red-400'
