   117→function KpiCard({
   118→  label,
   119→  value,
   120→  icon,
   121→  color,
   122→  subtitle,
   123→  onClick,
   124→  valueClassName,
   125→  pulse,
   126→}: {
   127→  label: string
   128→  value: React.ReactNode
   129→  icon: React.ReactNode
   130→  color: string
   131→  subtitle?: string
   132→  onClick?: () => void
   133→  valueClassName?: string
   134→  pulse?: boolean
   135→}) {
   136→  const c = colorMap[color] || colorMap.primary
   137→  return (
   138→    <Card
   139→      onClick={onClick}
   140→      className={cn(
   141→        'rounded-xl shadow-sm border-t-2 transition-all duration-200 hover:shadow-md hover:-translate-y-0.5 cursor-default hover:shadow-[inset_0_0_0_1px_rgba(0,0,0,0.05)] dark:hover:shadow-[inset_0_0_0_1px_rgba(255,255,255,0.05)]',
   142→        c.border,
   143→        c.gradient,
   144→        onClick && 'cursor-pointer',
   145→        pulse && 'pulse-glow'
   146→      )}
   147→    >
   148→      <CardContent className="p-5">
   149→        <div className="flex items-start justify-between">
   150→          <div className="space-y-1.5">
   151→            <p className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">

   153→            </p>
   154→            <p className={cn('text-2xl font-bold tracking-tight animate-count-up', valueClassName)}>
   155→              {value}
   156→            </p>
   157→            {subtitle && (
   158→              <p className="text-xs text-muted-foreground">{subtitle}</p>
   159→            )}
   160→          </div>

   163→              'h-11 w-11 rounded-xl flex items-center justify-center ring-1 shrink-0',

   167→          >
   168→            {icon}
   169→          </div>
   170→        </div>
   171→      </CardContent>
   172→    </Card>
   173→  )
   174→}
