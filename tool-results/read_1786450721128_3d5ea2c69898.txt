     3→import { useState, useMemo } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { useForm } from 'react-hook-form'
     6→import { zodResolver } from '@hookform/resolvers/zod'
     7→import { z } from 'zod'
     8→import { toast } from 'sonner'
     9→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    10→import {
    11→  Table,
    12→  TableBody,
    13→  TableCell,
    14→  TableHead,
    15→  TableHeader,
    16→  TableRow,
    17→} from '@/components/ui/table'
    18→import { Badge } from '@/components/ui/badge'
    19→import { Button } from '@/components/ui/button'
    20→import { Input } from '@/components/ui/input'
    21→import { Label } from '@/components/ui/label'
    22→import { Textarea } from '@/components/ui/textarea'
    23→import {
    24→  Select,
    25→  SelectContent,
    26→  SelectItem,
    27→  SelectTrigger,
    28→  SelectValue,
    29→} from '@/components/ui/select'
    30→import {
    31→  Dialog,
    32→  DialogContent,
    33→  DialogDescription,
    34→  DialogHeader,
    35→  DialogTitle,
    36→  DialogFooter,
    37→} from '@/components/ui/dialog'
    38→import {
    39→  AlertDialog,
    40→  AlertDialogAction,
    41→  AlertDialogCancel,
    42→  AlertDialogContent,
    43→  AlertDialogDescription,
    44→  AlertDialogFooter,
    45→  AlertDialogHeader,
    46→  AlertDialogTitle,
    47→} from '@/components/ui/alert-dialog'
    48→import { Skeleton } from '@/components/ui/skeleton'
    49→import { Plus, Search, Eye, Pencil, Trash2, Filter, Activity } from 'lucide-react'
    50→import { formatCurrency } from './lib/format'
    51→import {
    52→  AreaChart,
    53→  Area,
    54→  XAxis,
    55→  YAxis,
    56→  CartesianGrid,
    57→  Tooltip,
    58→  ResponsiveContainer,
    59→} from 'recharts'
    60→
    61→const productSchema = z.object({
    62→  sku: z.string().min(1, 'SKU requerido'),
    63→  nombre: z.string().min(1, 'Nombre requerido'),
    64→  descripcion: z.string().optional(),
    65→  idCategoria: z.coerce.number().nullable().optional(),
    66→  idMarca: z.coerce.number().nullable().optional(),
    67→  codigoBarras: z.string().optional(),
    68→  unidadMedida: z.string().default('unidad'),
    69→  costoUnitario: z.coerce.number().min(0),
    70→  precioVenta: z.coerce.number().min(0),
    71→  stockMinimo: z.coerce.number().min(0),
    72→  stockMaximo: z.coerce.number().nullable().optional(),
    73→  fotoUrl: z.string().optional(),
    74→  activo: z.boolean().default(true),
    75→})
    76→
    77→type ProductFormData = z.infer<typeof productSchema>
    78→
    79→interface Product {
    80→  id: number
    81→  sku: string
    82→  nombre: string
    83→  descripcion: string | null
    84→  idCategoria: number | null
    85→  idMarca: number | null
    86→  codigoBarras: string | null
    87→  unidadMedida: string
    88→  costoUnitario: number
    89→  precioVenta: number
    90→  stockMinimo: number
    91→  stockMaximo: number | null
    92→  fotoUrl: string | null
    93→  activo: boolean
    94→  fechaCreacion: string
    95→  categoria: { id: number; nombre: string } | null
    96→  marca: { id: number; nombre: string } | null
    97→  stocks: { idProducto: number; idUbicacion: number; cantidad: number }[]
    98→  totalStock: number
    99→}
   100→
   101→interface CategoriasResponse {
   102→  items?: Product[]
   103→  total: number
   104→}
   105→
   106→export function ProductsPage() {
   107→  const queryClient = useQueryClient()
   108→  const [page, setPage] = useState(1)
   109→  const [search, setSearch] = useState('')
   110→  const [searchInput, setSearchInput] = useState('')
   111→  const [idCategoria, setIdCategoria] = useState('')
   112→  const [idMarca, setIdMarca] = useState('')
   113→  const [bajoStock, setBajoStock] = useState(false)
   114→  const [showCreate, setShowCreate] = useState(false)
   115→  const [editId, setEditId] = useState<number | null>(null)
   116→  const [deleteId, setDeleteId] = useState<number | null>(null)
   117→  const [viewId, setViewId] = useState<number | null>(null)
   118→  const [timelineId, setTimelineId] = useState<number | null>(null)
   119→
   120→  const { data, isLoading } = useQuery<CategoriasResponse>({
   121→    queryKey: ['products', page, search, idCategoria, idMarca, bajoStock],
   122→    queryFn: () => {
   123→      const params = new URLSearchParams({
   124→        page: String(page),
   125→        pageSize: '20',
   126→      })
   127→      if (search) params.set('search', search)
   128→      if (idCategoria) params.set('categoria', idCategoria)
   129→      if (idMarca) params.set('marca', idMarca)
   130→      if (bajoStock) params.set('bajoStock', 'true')
   131→      return fetch(`/api/wms/productos?${params}`).then((r) => r.json())
   132→    },
   133→  })
   134→
   135→  const { data: viewProduct } = useQuery({
   136→    queryKey: ['product-detail', viewId],
   137→    queryFn: () => fetch(`/api/wms/productos/${viewId}`).then((r) => r.json()),
   138→    enabled: !!viewId,
   139→  })
   140→
   141→  const { data: timelineData } = useQuery({
   142→    queryKey: ['product-timeline', timelineId],
   143→    queryFn: () => fetch(`/api/wms/productos/${timelineId}/timeline`).then((r) => r.json()),
   144→    enabled: !!timelineId,
   145→  })
   146→
   147→  const { data: categorias = [] } = useQuery({
   148→    queryKey: ['categorias-list'],
   149→    queryFn: () => fetch('/api/wms/categorias').then((r) => r.json()),
   150→  })
   151→
   152→  const { data: marcas = [] } = useQuery({
   153→    queryKey: ['marcas-list'],
   154→    queryFn: () => fetch('/api/wms/marcas').then((r) => r.json()),
   155→  })
   156→
   157→  const form = useForm<ProductFormData>({
   158→    resolver: zodResolver(productSchema) as any,
   159→    defaultValues: {
   160→      sku: '',
   161→      nombre: '',
   162→      descripcion: '',
   163→      idCategoria: null,
   164→      idMarca: null,
   165→      codigoBarras: '',
   166→      unidadMedida: 'unidad',
   167→      costoUnitario: 0,
   168→      precioVenta: 0,
   169→      stockMinimo: 0,
   170→      stockMaximo: null,
   171→      fotoUrl: '',
   172→      activo: true,
   173→    },
   174→  })
   175→
   176→  const createMutation = useMutation({
   177→    mutationFn: (values: ProductFormData) =>
   178→      fetch('/api/wms/productos', {
   179→        method: 'POST',
   180→        headers: { 'Content-Type': 'application/json' },
   181→        body: JSON.stringify(values),
   182→      }).then((r) => {
   183→        if (!r.ok) return r.json().then((e) => Promise.reject(e))
   184→        return r.json()
   185→      }),
   186→    onSuccess: () => {
   187→      toast.success('Producto creado correctamente')
   188→      queryClient.invalidateQueries({ queryKey: ['products'] })
   189→      setShowCreate(false)
   190→      form.reset()
   191→    },
   192→    onError: (err: any) => {
   193→      toast.error(err.error ?? 'Error al crear producto')
   194→    },
   195→  })
   196→
   197→  const updateMutation = useMutation({
   198→    mutationFn: ({ id, values }: { id: number; values: ProductFormData }) =>
   199→      fetch(`/api/wms/productos/${id}`, {
   200→        method: 'PUT',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        body: JSON.stringify(values),
   203→      }).then((r) => {
   204→        if (!r.ok) return r.json().then((e) => Promise.reject(e))
   205→        return r.json()
   206→      }),
   207→    onSuccess: () => {
   208→      toast.success('Producto actualizado correctamente')
   209→      queryClient.invalidateQueries({ queryKey: ['products'] })
   210→      setEditId(null)
   211→      form.reset()
   212→    },
   213→    onError: (err: any) => {
   214→      toast.error(err.error ?? 'Error al actualizar producto')
   215→    },
   216→  })
   217→
   218→  const deleteMutation = useMutation({
   219→    mutationFn: (id: number) =>
   220→      fetch(`/api/wms/productos/${id}`, { method: 'DELETE' }).then((r) => {
   221→        if (!r.ok) return r.json().then((e) => Promise.reject(e))
   222→        return r.json()
   223→      }),
   224→    onSuccess: () => {
   225→      toast.success('Producto eliminado')
   226→      queryClient.invalidateQueries({ queryKey: ['products'] })
   227→      setDeleteId(null)
   228→    },
   229→    onError: (err: any) => {
   230→      toast.error(err.error ?? 'Error al eliminar producto')
   231→    },
   232→  })
   233→
   234→  function openEdit(product: Product) {
   235→    setEditId(product.id)
   236→    form.reset({
   237→      sku: product.sku,
   238→      nombre: product.nombre,
   239→      descripcion: product.descripcion ?? '',
   240→      idCategoria: product.idCategoria,
   241→      idMarca: product.idMarca,
   242→      codigoBarras: product.codigoBarras ?? '',
   243→      unidadMedida: product.unidadMedida,
   244→      costoUnitario: product.costoUnitario,
   245→      precioVenta: product.precioVenta,
   246→      stockMinimo: product.stockMinimo,
   247→      stockMaximo: product.stockMaximo,
   248→      fotoUrl: product.fotoUrl ?? '',
   249→      activo: product.activo,
   250→    })
   251→  }
   252→
   253→  function handleSearch() {
   254→    setSearch(searchInput)
   255→    setPage(1)
   256→  }
   257→
   258→  const items: Product[] = data?.items ?? []
   259→  const totalPages = Math.ceil((data?.total ?? 0) / 20)
   260→
   261→  function getStockStatus(p: Product) {
   262→    if (p.totalStock === 0) return 'out'
   263→    if (p.totalStock < p.stockMinimo) return 'low'
   264→    return 'ok'
   265→  }
   266→
   267→  function StatusBadge({ status }: { status: string }) {
   268→    if (status === 'out')
   269→      return <Badge variant="destructive" className="text-[10px]">Sin Stock</Badge>
   270→    if (status === 'low')
   271→      return (
   272→        <Badge className="bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-200 text-[10px]">
   273→          Bajo
   274→        </Badge>
   275→      )
   276→    return (
   277→      <Badge className="bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200 text-[10px]">
   278→        OK
   279→      </Badge>
   280→    )
   281→  }
   282→
   283→  const chartData = useMemo(() => {
   284→    if (!timelineData?.timeline) return []
   285→    return timelineData.timeline.map((t: any) => ({
   286→      fecha: new Date(t.date).toLocaleDateString('es-MX', { day: '2-digit', month: 'short' }),
   287→      stock: t.balance,
   288→    }))
   289→  }, [timelineData])
   290→
   291→  return (
   292→    <div className="space-y-4">
   293→      {/* Toolbar */}
   294→      <div className="flex flex-col sm:flex-row gap-3">
   295→        <div className="flex gap-2 flex-1 flex-wrap">
   296→          <div className="relative flex-1 max-w-sm">
   297→            <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   298→            <Input
   299→              placeholder="Buscar por nombre o SKU..."
   300→              className="pl-9"
   301→              value={searchInput}
   302→              onChange={(e) => setSearchInput(e.target.value)}
   303→              onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   304→            />
   305→          </div>
   306→          <Button variant="outline" onClick={handleSearch}>
   307→            <Search className="h-4 w-4" />
   308→          </Button>
   309→          <Button
   310→            variant={bajoStock ? 'default' : 'outline'}
   311→            onClick={() => { setBajoStock(!bajoStock); setPage(1) }}
   312→          >
   313→            <Filter className="h-4 w-4 mr-1" />
   314→            Bajo Stock
   315→          </Button>
   316→          <Select value={idCategoria} onValueChange={(v) => { setIdCategoria(v === '_all' ? '' : v); setPage(1) }}>
   317→            <SelectTrigger className="w-[140px] h-9 text-xs">
   318→              <SelectValue placeholder="Categoría" />
   319→            </SelectTrigger>
   320→            <SelectContent>
   321→              <SelectItem value="_all">Todas</SelectItem>
   322→              {categorias.map((c: any) => (
   323→                <SelectItem key={c.id} value={String(c.id)}>{c.nombre}</SelectItem>
   324→              ))}
   325→            </SelectContent>
   326→          </Select>
   327→          <Select value={idMarca} onValueChange={(v) => { setIdMarca(v === '_all' ? '' : v); setPage(1) }}>
   328→            <SelectTrigger className="w-[140px] h-9 text-xs">
   329→              <SelectValue placeholder="Marca" />
   330→            </SelectTrigger>
   331→            <SelectContent>
   332→              <SelectItem value="_all">Todas</SelectItem>
   333→              {marcas.map((m: any) => (
   334→                <SelectItem key={m.id} value={String(m.id)}>{m.nombre}</SelectItem>
   335→              ))}
   336→            </SelectContent>
   337→          </Select>
   338→        </div>
   339→        <Button onClick={() => { form.reset(); setShowCreate(true) }}>
   340→          <Plus className="h-4 w-4 mr-1" />
   341→          Nuevo Producto
   342→        </Button>
   343→      </div>
   344→
   345→      <div className="mb-4">
   346→        <p className="text-sm text-muted-foreground">Gestiona el catálogo de repuestos, equipos y compatibilidades</p>
   347→      </div>
   348→
   349→      {/* Table */}
   350→      <Card className="rounded-xl shadow-sm transition-all duration-200">
   351→        <CardContent className="p-0">
   352→          <div className="table-container max-h-[calc(100vh-14rem)] overflow-y-auto overflow-x-auto">
   353→            <Table>
   354→              <TableHeader>
   355→                <TableRow>
   356→                  <TableHead className="text-xs">SKU</TableHead>
   357→                  <TableHead className="text-xs">Nombre</TableHead>
   358→                  <TableHead className="text-xs hidden md:table-cell">Categoría</TableHead>
   359→                  <TableHead className="text-xs hidden lg:table-cell">Marca</TableHead>
   360→                  <TableHead className="text-xs text-right hidden sm:table-cell">Costo</TableHead>
   361→                  <TableHead className="text-xs text-right">Precio</TableHead>
   362→                  <TableHead className="text-xs text-center">Stock</TableHead>
   363→                  <TableHead className="text-xs text-center">Estado</TableHead>
   364→                  <TableHead className="text-xs text-right">Acciones</TableHead>
   365→                </TableRow>
   366→              </TableHeader>
   367→              <TableBody>
   368→                {isLoading &&
   369→                  Array.from({ length: 5 }).map((_, i) => (
   370→                    <TableRow key={i}>
   371→                      {Array.from({ length: 9 }).map((_, j) => (
   372→                        <TableCell key={j} className="py-2">
   373→                          <Skeleton className="h-4 w-full" />
   374→                        </TableCell>
   375→                      ))}
   376→                    </TableRow>
   377→                  ))}
   378→                {!isLoading && items.length === 0 && (
   379→                  <TableRow>
   380→                    <TableCell colSpan={9} className="text-center text-muted-foreground py-8">
   381→                      No se encontraron productos
   382→                    </TableCell>
   383→                  </TableRow>
   384→                )}
   385→                {!isLoading &&
   386→                  items.map((p) => (
   387→                    <TableRow key={p.id} className="hover:bg-muted/50">
   388→                      <TableCell className="text-xs font-mono py-2">{p.sku}</TableCell>
   389→                      <TableCell className="text-xs font-medium py-2 max-w-[200px] truncate">
   390→                        {p.nombre}
   391→                      </TableCell>
   392→                      <TableCell className="text-xs py-2 hidden md:table-cell">
   393→                        {p.categoria?.nombre ?? '-'}
   394→                      </TableCell>
   395→                      <TableCell className="text-xs py-2 hidden lg:table-cell">
   396→                        {p.marca?.nombre ?? '-'}
   397→                      </TableCell>
   398→                      <TableCell className="text-xs text-right py-2 hidden sm:table-cell">
   399→                        {formatCurrency(p.costoUnitario)}
   400→                      </TableCell>
   401→                      <TableCell className="text-xs text-right py-2">
   402→                        {formatCurrency(p.precioVenta)}
   403→                      </TableCell>
   404→                      <TableCell className="text-xs text-center font-mono py-2">
   405→                        {p.totalStock}
   406→                      </TableCell>
   407→                      <TableCell className="text-xs text-center py-2">
   408→                        <StatusBadge status={getStockStatus(p)} />
   409→                      </TableCell>
   410→                      <TableCell className="text-xs text-right py-2">
   411→                        <div className="flex justify-end gap-1">
   412→                          <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => setViewId(p.id)}>
   413→                            <Eye className="h-3.5 w-3.5" />
   414→                          </Button>
   415→                          <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => openEdit(p)}>
   416→                            <Pencil className="h-3.5 w-3.5" />
   417→                          </Button>
   418→                          <Button size="icon" variant="ghost" className="h-7 w-7 text-destructive" onClick={() => setDeleteId(p.id)}>
   419→                            <Trash2 className="h-3.5 w-3.5" />
   420→                          </Button>
   421→                        </div>
   422→                      </TableCell>
   423→                    </TableRow>
   424→                  ))}
   425→              </TableBody>
   426→            </Table>
   427→          </div>
   428→        </CardContent>
   429→      </Card>
   430→
   431→      {/* Pagination */}
   432→      <div className="flex items-center justify-between text-sm">
   433→        <p className="text-muted-foreground">
   434→          {(data?.total ?? 0)} productos encontrados
   435→        </p>
   436→        <div className="flex gap-2">
   437→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(page - 1)}>
   438→            Anterior
   439→          </Button>
   440→          <span className="px-3 py-1 text-sm">{page} / {totalPages || 1}</span>
   441→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>
   442→            Siguiente
   443→          </Button>
   444→        </div>
   445→      </div>
   446→
   447→      {/* Create/Edit Dialog */}
   448→      <Dialog open={showCreate || !!editId} onOpenChange={(open) => { if (!open) { setShowCreate(false); setEditId(null); form.reset() } }}>
   449→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   450→          <DialogHeader>
   451→            <DialogTitle>{editId ? 'Editar Producto' : 'Nuevo Producto'}</DialogTitle>
   452→            <DialogDescription className="sr-only">{editId ? 'Formulario para editar los datos del producto' : 'Formulario para crear un nuevo producto'}</DialogDescription>
   453→          </DialogHeader>
   454→          <form
   455→            onSubmit={form.handleSubmit((values: any) => {
   456→              if (editId) {
   457→                updateMutation.mutate({ id: editId, values })
   458→              } else {
   459→                createMutation.mutate(values)
   460→              }
   461→            })}
   462→            className="space-y-4"
   463→          >
   464→            <div className="grid grid-cols-2 gap-4">
   465→              <div className="space-y-2">
   466→                <Label className="text-sm font-medium">SKU *</Label>
   467→                <Input {...form.register('sku')} />
   468→                {form.formState.errors.sku && (
   469→                  <p className="text-xs text-destructive">{form.formState.errors.sku.message}</p>
   470→                )}
   471→              </div>
   472→              <div className="space-y-2">
   473→                <Label className="text-sm font-medium">Unidad de Medida</Label>
   474→                <Select value={form.watch('unidadMedida')} onValueChange={(v) => form.setValue('unidadMedida', v)}>
   475→                  <SelectTrigger><SelectValue /></SelectTrigger>
   476→                  <SelectContent>
   477→                    <SelectItem value="unidad">Unidad</SelectItem>
   478→                    <SelectItem value="par">Par</SelectItem>
   479→                    <SelectItem value="metro">Metro</SelectItem>
   480→                    <SelectItem value="kg">Kg</SelectItem>
   481→                    <SelectItem value="litro">Litro</SelectItem>
   482→                  </SelectContent>
   483→                </Select>
   484→              </div>
   485→            </div>
   486→            <div className="space-y-2">
   487→              <Label className="text-sm font-medium">Nombre *</Label>
   488→              <Input {...form.register('nombre')} />
   489→              {form.formState.errors.nombre && (
   490→                <p className="text-xs text-destructive">{form.formState.errors.nombre.message}</p>
   491→              )}
   492→            </div>
   493→            <div className="space-y-2">
   494→              <Label className="text-sm font-medium">Descripción</Label>
   495→              <Textarea {...form.register('descripcion')} rows={2} />
   496→            </div>
   497→            <div className="grid grid-cols-2 gap-4">
   498→              <div className="space-y-2">
   499→                <Label className="text-sm font-medium">Código de Barras</Label>
   500→                <Input {...form.register('codigoBarras')} />
   501→              </div>
   502→              <div className="space-y-2">
   503→                <Label className="text-sm font-medium">URL de Foto</Label>
   504→                <Input {...form.register('fotoUrl')} />
   505→              </div>
   506→            </div>
   507→            {form.watch('fotoUrl') && (
   508→              <div className="mt-2">
   509→                <img
   510→                  src={form.watch('fotoUrl')}
   511→                  alt="Preview"
   512→                  className="w-24 h-24 object-cover rounded-lg border"
   513→                  onError={(e) => { (e.target as HTMLImageElement).style.display = 'none' }}
   514→                />
   515→              </div>
   516→            )}
   517→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   518→              <div className="space-y-2">
   519→                <Label className="text-sm font-medium">Costo Unitario</Label>
   520→                <Input type="number" step="0.01" {...form.register('costoUnitario')} />
   521→              </div>
   522→              <div className="space-y-2">
   523→                <Label className="text-sm font-medium">Precio de Venta</Label>
   524→                <Input type="number" step="0.01" {...form.register('precioVenta')} />
   525→              </div>
   526→              <div className="space-y-2">
   527→                <Label className="text-sm font-medium">Stock Mínimo</Label>

   584→                  <p className="font-medium mb-2">Stock por Ubicación</p>

   602→                    ))}
   603→                  </div>

   607→                <Button
   608→                  variant="outline"
   609→                  size="sm"
   610→                  onClick={() => { setTimelineId(viewProduct.id); setViewId(null) }}
   611→                >
   612→                  <Activity className="h-4 w-4 mr-1" />
   613→                  Ver Timeline
   614→                </Button>
   615→              </div>
   616→            </div>
   617→          )}
   618→        </DialogContent>
   619→      </Dialog>
   620→
   621→      {/* Timeline Dialog */}
   622→      <Dialog open={!!timelineId} onOpenChange={(open) => { if (!open) setTimelineId(null) }}>
   623→        <DialogContent className="max-w-xl max-h-[90vh] overflow-y-auto">
   624→          <DialogHeader>
   625→            <DialogTitle>Timeline de Stock</DialogTitle>

   627→          </DialogHeader>
   628→          {timelineData && (
   629→            <div className="space-y-4">
   630→              <div className="flex items-center justify-between text-sm">

   639→              </div>
   640→              {chartData.length > 0 ? (

   648→                        </linearGradient>
   649→                      </defs>
   650→                      <CartesianGrid strokeDasharray="3 3" className="stroke-border" />
   651→                      <XAxis dataKey="fecha" tick={{ fontSize: 11 }} />
   652→                      <YAxis tick={{ fontSize: 11 }} />
   653→                      <Tooltip
   654→                        contentStyle={{
   655→                          borderRadius: '8px',
   656→                          border: '1px solid var(--border)',
   657→                          background: 'var(--popover)',
   658→                          color: 'var(--popover-foreground)',
   659→                        }}
   660→                      />
   661→                      <Area
   662→                        type="stepAfter"
   663→                        dataKey="stock"
   664→                        stroke="oklch(0.55 0.15 145)"

   667→                      />

   685→            <AlertDialogTitle>¿Eliminar producto?</AlertDialogTitle>

   687→              Esta acción no se puede deshacer. El producto será eliminado permanentemente.
   688→            </AlertDialogDescription>
   689→          </AlertDialogHeader>
   690→          <AlertDialogFooter>
   691→            <AlertDialogCancel>Cancelar</AlertDialogCancel>
   692→            <AlertDialogAction onClick={() => deleteId && deleteMutation.mutate(deleteId)}>
   693→              Eliminar
   694→            </AlertDialogAction>
   695→          </AlertDialogFooter>
   696→        </AlertDialogContent>
   697→      </AlertDialog>
   698→    </div>
   699→  )
