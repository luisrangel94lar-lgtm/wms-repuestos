   108→function MarginCalculator({ costoUnitario, precioVenta }: { costoUnitario: number; precioVenta: number }) {
   109→  const margen = precioVenta > 0 ? ((precioVenta - costoUnitario) / precioVenta) * 100 : 0
   110→  const utilidad = precioVenta - costoUnitario
   111→  const clampedMargin = Math.max(0, Math.min(100, margen))
   112→
   113→  let barColor = 'bg-red-500'
   114→  let textColor = 'text-red-600 dark:text-red-400'
   115→  if (margen > 40) {
   116→    barColor = 'bg-emerald-500'
   117→    textColor = 'text-emerald-600 dark:text-emerald-400'
   118→  } else if (margen >= 20) {
   119→    barColor = 'bg-amber-500'
   120→    textColor = 'text-amber-600 dark:text-amber-400'
   121→  }
   122→
   123→  return (
   124→    <div className="rounded-lg border bg-muted/30 p-4 space-y-3">
   125→      <div className="flex items-center gap-2 text-sm font-medium">
   126→        <Calculator className="h-4 w-4 text-primary" />
   127→        Calculadora de Margen
   128→      </div>
   129→      <div className="grid grid-cols-3 gap-4 text-center">
   130→        <div>
   131→          <p className="text-xs text-muted-foreground">Costo</p>
   132→          <p className="text-sm font-mono font-semibold">{formatCurrency(costoUnitario)}</p>
   133→        </div>
   134→        <div>
   135→          <p className="text-xs text-muted-foreground">Precio Venta</p>
   136→          <p className="text-sm font-mono font-semibold">{formatCurrency(precioVenta)}</p>
   137→        </div>
   138→        <div>
   139→          <p className="text-xs text-muted-foreground">Utilidad</p>
   140→          <p className={cn('text-sm font-mono font-bold', utilidad > 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400')}>
   141→            {formatCurrency(utilidad)}
   142→          </p>
   143→        </div>
   144→      </div>
   145→      <div className="space-y-1.5">
   146→        <div className="flex items-center justify-between text-xs">
   147→          <span className="text-muted-foreground">Margen</span>
   148→          <span className={cn('font-bold font-mono', textColor)}>{margen.toFixed(1)}%</span>
   149→        </div>
   150→        <div className="h-2.5 w-full bg-muted rounded-full overflow-hidden">
   151→          <div
   152→            className={cn('h-full rounded-full transition-all duration-300', barColor)}
   153→            style={{ width: `${clampedMargin}%` }}
   154→          />
   155→        </div>
   156→        <div className="flex justify-between text-[10px] text-muted-foreground">
   157→          <span>Bajo (&lt;20%)</span>
   158→          <span>Medio (20-40%)</span>
   159→          <span>Alto (&gt;40%)</span>
   160→        </div>
   161→      </div>
   162→    </div>
   163→  )
   164→}
