   109→function MarginCalculator({ costoUnitario, precioVenta }: { costoUnitario: number; precioVenta: number }) {
   110→  const margen = precioVenta > 0 ? ((precioVenta - costoUnitario) / precioVenta) * 100 : 0
   111→  const utilidad = precioVenta - costoUnitario
   112→  const clampedMargin = Math.max(0, Math.min(100, margen))
   113→
   114→  let barColor = 'bg-red-500'
   115→  let textColor = 'text-red-600 dark:text-red-400'
   116→  if (margen > 40) {
   117→    barColor = 'bg-emerald-500'
   118→    textColor = 'text-emerald-600 dark:text-emerald-400'
   119→  } else if (margen >= 20) {
   120→    barColor = 'bg-amber-500'
   121→    textColor = 'text-amber-600 dark:text-amber-400'
   122→  }
   123→
   124→  return (
   125→    <div className="rounded-lg border bg-muted/30 p-4 space-y-3">
   126→      <div className="flex items-center gap-2 text-sm font-medium">
   127→        <Calculator className="h-4 w-4 text-primary" />
   128→        Calculadora de Margen
   129→      </div>
   130→      <div className="grid grid-cols-3 gap-4 text-center">
   131→        <div>
   132→          <p className="text-xs text-muted-foreground">Costo</p>
   133→          <p className="text-sm font-mono font-semibold">{formatCurrency(costoUnitario)}</p>
   134→        </div>
   135→        <div>
   136→          <p className="text-xs text-muted-foreground">Precio Venta</p>
   137→          <p className="text-sm font-mono font-semibold">{formatCurrency(precioVenta)}</p>
   138→        </div>
   139→        <div>
   140→          <p className="text-xs text-muted-foreground">Utilidad</p>
   141→          <p className={cn('text-sm font-mono font-bold', utilidad > 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400')}>
   142→            {formatCurrency(utilidad)}
   143→          </p>
   144→        </div>
   145→      </div>
   146→      <div className="space-y-1.5">
   147→        <div className="flex items-center justify-between text-xs">
   148→          <span className="text-muted-foreground">Margen</span>
   149→          <span className={cn('font-bold font-mono', textColor)}>{margen.toFixed(1)}%</span>
   150→        </div>
   151→        <div className="h-2.5 w-full bg-muted rounded-full overflow-hidden">
   152→          <div
   153→            className={cn('h-full rounded-full transition-all duration-300', barColor)}
   154→            style={{ width: `${clampedMargin}%` }}
   155→          />
   156→        </div>
   157→        <div className="flex justify-between text-[10px] text-muted-foreground">
   158→          <span>Bajo (&lt;20%)</span>
   159→          <span>Medio (20-40%)</span>
   160→          <span>Alto (&gt;40%)</span>
   161→        </div>
   162→      </div>
   163→    </div>
   164→  )
   165→}
