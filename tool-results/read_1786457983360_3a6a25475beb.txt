   119→function KpiCard({
   120→  label,
   121→  value,
   122→  icon,
   123→  color,
   124→  subtitle,
   125→  onClick,
   126→  valueClassName,
   127→  pulse,
   128→  trend,
   129→}: {
   130→  label: string
   131→  value: React.ReactNode
   132→  icon: React.ReactNode
   133→  color: string
   134→  subtitle?: string
   135→  onClick?: () => void
   136→  valueClassName?: string
   137→  pulse?: boolean
   138→  trend?: number
   139→}) {
   140→  const c = colorMap[color] || colorMap.primary
   141→  return (
   142→    <Card
   143→      onClick={onClick}
   144→      className={cn(
   145→        'rounded-xl shadow-sm border-t-2 transition-all duration-200 hover:shadow-md hover:-translate-y-0.5 cursor-default hover:shadow-[inset_0_0_0_1px_rgba(0,0,0,0.05)] dark:hover:shadow-[inset_0_0_0_1px_rgba(255,255,255,0.05)]',
   146→        c.border,
   147→        c.gradient,
   148→        onClick && 'cursor-pointer',
   149→        pulse && 'pulse-glow'
   150→      )}
   151→    >
   152→      <CardContent className="p-5">
   153→        <div className="flex items-start justify-between">
   154→          <div className="space-y-1.5">
   155→            <p className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   156→              {label}
   157→            </p>
   158→            <p className={cn('text-2xl font-bold tracking-tight animate-count-up stat-number', valueClassName)}>
   159→              {value}
   160→            </p>
   161→            {subtitle && (
   162→              <p className="text-xs text-muted-foreground">{subtitle}</p>
   163→            )}
   164→            {trend !== undefined && trend !== 0 && (
   165→              <p className={cn(
   166→                'text-[11px] font-medium flex items-center gap-0.5',
   167→                trend > 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400'
   168→              )}>
   169→                {trend > 0 ? <ArrowUp className="h-3 w-3" /> : <ArrowDown className="h-3 w-3" />}
   170→                {Math.abs(trend)}%
   171→              </p>
   172→            )}
   173→          </div>
   174→          <div
   175→            className={cn(
   176→              'h-11 w-11 rounded-xl flex items-center justify-center ring-1 shrink-0',
   177→              c.iconBg,
   178→              c.ring
   179→            )}
   180→          >
   181→            {icon}
   182→          </div>
   183→        </div>
   184→      </CardContent>
   185→    </Card>
   186→  )
   187→}
