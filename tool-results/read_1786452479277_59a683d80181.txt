    35→interface StockEntry {
    36→  idProducto: number
    37→  idUbicacion: number
    38→  cantidad: number
    39→  producto: { id: number; sku: string; nombre: string; costoUnitario: number; precioVenta: number; stockMinimo: number }
    40→  ubicacion: { id: number; pasillo: string; estante: string; nivel: string }
    41→}
    42→
    43→interface InventoryItem {
    44→  id: number
    45→  sku: string
    46→  nombre: string
    47→  descripcion: string | null
    48→  idCategoria: number | null
    49→  idMarca: number | null
    50→  costoUnitario: number
    51→  precioVenta: number
    52→  stockMinimo: number
    53→  totalStock: number
    54→  valorTotal: number
    55→  status: 'ok' | 'low' | 'out'
    56→  stocks: StockEntry[]
    57→  categoria?: { id: number; nombre: string } | null
    58→  marca?: { id: number; nombre: string } | null
    59→}
    60→
    61→function StatusBadge({ status }: { status: string }) {
    62→  if (status === 'out')
    63→    return (
    64→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-danger">
    65→        <XCircle className="h-3 w-3" /> Sin Stock
    66→      </span>
    67→    )
    68→  if (status === 'low')
    69→    return (
    70→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-warning">
    71→        <AlertTriangle className="h-3 w-3" /> Bajo Stock
    72→      </span>
    73→    )
    74→  return (
    75→    <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-success">
    76→      <Check className="h-3 w-3" /> OK
    77→    </span>
    78→  )
    79→}
    80→
    81→export function InventoryPage() {
    82→  const queryClient = useQueryClient()
    83→  const { setCurrentPage, setReceivingProductId } = useWmsStore()
    84→  const [selectedProduct, setSelectedProduct] = useState<StockEntry['producto'] | null>(null)
    85→  const [sheetProduct, setSheetProduct] = useState<InventoryItem | null>(null)
    86→
    87→  // Ajuste state
    88→  const [ajusteProduct, setAjusteProduct] = useState<InventoryItem | null>(null)
    89→  const [ajusteStock, setAjusteStock] = useState('')
    90→  const [ajusteMotivo, setAjusteMotivo] = useState('')
    91→  const [ajusteLoading, setAjusteLoading] = useState(false)
    92→
    93→  // Traslado state
    94→  const [trasladoProduct, setTrasladoProduct] = useState<InventoryItem | null>(null)
    95→  const [trasladoOrigen, setTrasladoOrigen] = useState('')
    96→  const [trasladoDestino, setTrasladoDestino] = useState('')
    97→  const [trasladoCantidad, setTrasladoCantidad] = useState('')
    98→  const [trasladoLoading, setTrasladoLoading] = useState(false)
    99→
   100→  // Sorting state
   101→  const [sortField, setSortField] = useState<string>('nombre')
   102→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   103→
   104→  const { data: products = [], isLoading } = useQuery({
   105→    queryKey: ['products-inventory'],
   106→    queryFn: () => fetch('/api/wms/productos?pageSize=500').then((r) => r.json()).then((d: any) => d.items ?? []),
   107→  })
   108→
   109→  const { data: stockEntries = [] } = useQuery<StockEntry[]>({
   110→    queryKey: ['all-stock-inv'],
   111→    queryFn: () => fetch('/api/wms/stock').then((r) => r.json()),
   112→  })
   113→
   114→  const { data: allLocations = [] } = useQuery({
   115→    queryKey: ['all-locations'],
   116→    queryFn: () => fetch('/api/wms/ubicaciones?activos=true').then((r) => r.json()),
   117→  })
   118→
   119→  const { data: productMovements = [] } = useQuery({
   120→    queryKey: ['product-movements-sheet', sheetProduct?.id],
   121→    queryFn: () => fetch(`/api/wms/movimientos?idProducto=${sheetProduct?.id}&limit=5`).then((r) => r.json()),
   122→    enabled: !!sheetProduct?.id,
   123→  })
   124→
   125→  // Build inventory view
   126→  const inventory: InventoryItem[] = products.map((p: any) => {
   127→    const stocks = stockEntries.filter((s) => s.idProducto === p.id)
   128→    const totalStock = stocks.reduce((sum, s) => sum + s.cantidad, 0)
   129→    const valorTotal = totalStock * p.costoUnitario
   130→    let status: 'ok' | 'low' | 'out' = 'ok'
   131→    if (totalStock === 0) status = 'out'
   132→    else if (totalStock < p.stockMinimo) status = 'low'
   133→    return { ...p, totalStock, valorTotal, status, stocks }
   134→  })
   135→
   136→  const { data: inventoryStats } = useQuery({
   137→    queryKey: ['inventory-stats'],
   138→    queryFn: () => fetch('/api/wms/dashboard').then(r => r.json()).then(d => ({
   139→      totalProducts: d.totalProductos,
   140→      stockValue: d.valorTotalStock,
   141→      bajoStock: d.productosBajoStock?.count ?? 0,
   142→    })),
   143→  })
   144→
   145→  const totalValor = inventory.reduce((sum, p) => sum + p.valorTotal, 0)
   146→
   147→  function invalidateInventory() {
   148→    queryClient.invalidateQueries({ queryKey: ['products-inventory'] })
   149→    queryClient.invalidateQueries({ queryKey: ['all-stock-inv'] })
   150→    queryClient.invalidateQueries({ queryKey: ['inventory-stats'] })
   151→    queryClient.invalidateQueries({ queryKey: ['product-movements-sheet'] })
   152→  }
   153→
   154→  // --- Sorting ---
   155→  function handleSort(field: string) {
   156→    if (sortField === field) {
   157→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   158→    } else {
   159→      setSortField(field)
   160→      setSortDir('asc')
   161→    }
   162→  }
   163→  function getSortValue(item: InventoryItem, field: string): any {
   164→    if (field === 'margen') {
   165→      return item.precioVenta > 0 ? ((item.precioVenta - item.costoUnitario) / item.precioVenta * 100) : 0
   166→    }
   167→    return item[field as keyof InventoryItem]
   168→  }
   169→
   170→  const sortedInventory = [...inventory].sort((a, b) => {
   171→    let aVal: any = getSortValue(a, sortField)
   172→    let bVal: any = getSortValue(b, sortField)
   173→    if (typeof aVal === 'string') { aVal = aVal.toLowerCase(); bVal = (bVal as string).toLowerCase() }
   174→    if (aVal < bVal) return sortDir === 'asc' ? -1 : 1
   175→    if (aVal > bVal) return sortDir === 'asc' ? 1 : -1
   176→    return 0
   177→  })
   178→
   179→  // --- AJUSTE handlers ---
   180→  function openAjuste(item: InventoryItem) {
   181→    setAjusteProduct(item)
   182→    setAjusteStock(String(item.totalStock))
   183→    setAjusteMotivo('')
   184→  }
   185→
   186→  async function submitAjuste() {
   187→    if (!ajusteProduct || ajusteStock === '') return
   188→    const newStock = parseInt(ajusteStock, 10)
   189→    if (isNaN(newStock) || newStock < 0) {
   190→      toast.error('Cantidad inválida')
   191→      return
   192→    }
   193→
   194→    const locationId = ajusteProduct.stocks.length > 0
   195→      ? ajusteProduct.stocks[0].idUbicacion
   196→      : allLocations[0]?.id
   197→
   198→    if (!locationId) {
   199→      toast.error('No hay ubicaciones disponibles')
   200→      return
   201→    }
   202→
   203→    setAjusteLoading(true)
   204→    try {
   205→      const res = await fetch('/api/wms/movimientos', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({
   209→          idProducto: ajusteProduct.id,
   210→          idUbicacion: locationId,
   211→          idTipo: 3,
   212→          cantidad: newStock,
   213→          referencia: 'Ajuste manual',
   214→          observacion: ajusteMotivo || null,
   215→        }),
   216→      })
   217→      if (!res.ok) {
   218→        const err = await res.json()
   219→        throw new Error(err.error || 'Error al ajustar stock')
   220→      }
   221→      toast.success('Stock ajustado correctamente')
   222→      setAjusteProduct(null)
   223→      invalidateInventory()
   224→    } catch (e: any) {
   225→      toast.error(e.message || 'Error al ajustar stock')
   226→    } finally {
   227→      setAjusteLoading(false)
   228→    }
   229→  }
   230→
   231→  // --- TRASLADO handlers ---
   232→  function openTraslado(item: InventoryItem) {
   233→    if (item.stocks.length < 1) return
   234→    setTrasladoProduct(item)
   235→    setTrasladoOrigen(String(item.stocks[0].idUbicacion))
   236→    setTrasladoDestino('')
   237→    setTrasladoCantidad('')
   238→  }
   239→
   240→  function getOrigenStock(): number {
   241→    if (!trasladoProduct) return 0
   242→    const entry = trasladoProduct.stocks.find((s) => String(s.idUbicacion) === trasladoOrigen)
   243→    return entry?.cantidad ?? 0
   244→  }
   245→
   246→  async function submitTraslado() {
   247→    if (!trasladoProduct || !trasladoOrigen || !trasladoDestino || !trasladoCantidad) return
   248→    if (trasladoOrigen === trasladoDestino) {
   249→      toast.error('El origen y destino no pueden ser iguales')
   250→      return
   251→    }
   252→    const qty = parseInt(trasladoCantidad, 10)
   253→    if (isNaN(qty) || qty <= 0) {
   254→      toast.error('Cantidad inválida')
   255→      return
   256→    }
   257→    const origenStock = getOrigenStock()
   258→    if (qty > origenStock) {
   259→      toast.error(`Stock insuficiente en origen (disponible: ${origenStock})`)
   260→      return
   261→    }
   262→
   263→    setTrasladoLoading(true)
   264→    try {
   265→      const res = await fetch('/api/wms/stock/transfer', {
   266→        method: 'POST',
   267→        headers: { 'Content-Type': 'application/json' },
   268→        body: JSON.stringify({
   269→          idProducto: trasladoProduct.id,
   270→          idUbicacionOrigen: parseInt(trasladoOrigen, 10),
   271→          idUbicacionDestino: parseInt(trasladoDestino, 10),
   272→          cantidad: qty,
   273→        }),
   274→      })
   275→      if (!res.ok) {
   276→        const err = await res.json()
   277→        throw new Error(err.error || 'Error en traslado')
   278→      }
   279→      toast.success('Traslado realizado correctamente')
   280→      setTrasladoProduct(null)
   281→      invalidateInventory()
   282→    } catch (e: any) {
   283→      toast.error(e.message || 'Error en traslado')
   284→    } finally {
   285→      setTrasladoLoading(false)
   286→    }
   287→  }
   288→
   289→  // --- CSV Export ---
   290→  function exportInventory() {
   291→    exportToCSV(
   292→      inventory.map((p) => ({
   293→        producto: p.nombre,
   294→        sku: p.sku,
   295→        totalStock: p.totalStock,
   296→        valorTotal: p.valorTotal,
   297→      })),
   298→      'inventario',
   299→      [
   300→        { key: 'producto', label: 'Producto' },
   301→        { key: 'sku', label: 'SKU' },
   302→        { key: 'totalStock', label: 'Stock Total' },
   303→        { key: 'valorTotal', label: 'Valor Total', format: 'currency' },
   304→      ],
   305→    )
   306→  }
   307→
   308→  function openSheet(item: InventoryItem) {
   309→    setSheetProduct(item)
   310→  }
   311→
   312→  function quickGoToReceiving() {
   313→    if (!sheetProduct) return
   314→    const deficiency = sheetProduct.stockMinimo - sheetProduct.totalStock
   315→    setReceivingProductId(sheetProduct.id, deficiency > 0 ? deficiency : null)
   316→    setCurrentPage('receiving')
   317→    setSheetProduct(null)
   318→  }
   319→
   320→  function quickAdjust() {
   321→    if (!sheetProduct) return
   322→    setSheetProduct(null)
   323→    openAjuste(sheetProduct)
   324→  }
   325→
   326→  function quickTransfer() {
   327→    if (!sheetProduct) return
   328→    setSheetProduct(null)
   329→    openTraslado(sheetProduct)
   330→  }
   331→
   332→  // Find max stock for the location bars
   333→  function getMaxStock(): number {
   334→    if (!sheetProduct) return 1
   335→    const max = Math.max(...sheetProduct.stocks.map((s) => s.cantidad), sheetProduct.stockMinimo)
   336→    return max || 1
   337→  }
   338→
   339→  return (
   340→    <div className="space-y-4">
   341→      {/* Stats Bar */}
   342→      <div className="flex flex-wrap gap-3 mb-4">
   343→        <Card className="rounded-lg px-4 py-3 shadow-sm">
   344→          <p className="text-xs text-muted-foreground">Productos Activos</p>
   345→          <p className="text-lg font-bold">{products.length}</p>
   346→        </Card>
   347→        <Card className="rounded-lg px-4 py-3 shadow-sm">
   348→          <p className="text-xs text-muted-foreground">Valor Total</p>
   349→          <p className="text-lg font-bold">{formatCurrency(inventoryStats?.stockValue ?? totalValor)}</p>
   350→        </Card>
   351→        <Card className="rounded-lg px-4 py-3 shadow-sm">
   352→          <p className="text-xs text-muted-foreground">Bajo Mínimo</p>
   353→          <p className="text-lg font-bold text-destructive">{inventoryStats?.bajoStock ?? 0}</p>
   354→        </Card>
   355→      </div>
   356→
   357→      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   358→        <div>
   359→          <p className="text-sm text-muted-foreground">
   360→            {inventory.length} productos · Valor total: <span className="font-semibold text-foreground">{formatCurrency(totalValor)}</span>
   361→          </p>
   362→        </div>
   363→        <Button variant="outline" onClick={exportInventory}>
   364→          <Download className="h-4 w-4 mr-1" /> Exportar CSV
   365→        </Button>
   366→      </div>
   367→
   368→      <div className="mb-4">
   369→        <p className="text-sm text-muted-foreground">Vista consolidada del inventario actual</p>
   370→      </div>
   371→
   372→      <Card className="rounded-xl shadow-sm">
   373→        <CardContent className="p-0">
   374→          <div className="table-container max-h-[calc(100vh-14rem)] overflow-y-auto">
   375→            <Table>
   376→              <TableHeader>
   377→                <TableRow>
   378→                  <SortableHeader field="nombre" sortField={sortField} sortDir={sortDir} onSort={handleSort}>Producto</SortableHeader>
   379→                  <SortableHeader field="sku" sortField={sortField} sortDir={sortDir} onSort={handleSort}>SKU</SortableHeader>
   380→                  <SortableHeader field="totalStock" align="center" sortField={sortField} sortDir={sortDir} onSort={handleSort}>Total Stock</SortableHeader>
   381→                  <SortableHeader field="valorTotal" align="right" sortField={sortField} sortDir={sortDir} onSort={handleSort}>Valor Total</SortableHeader>
   382→                  <SortableHeader field="margen" align="right" sortField={sortField} sortDir={sortDir} onSort={handleSort}>Margen %</SortableHeader>
   383→                  <TableHead className="text-xs text-center">Estado</TableHead>
   384→                  <TableHead className="text-xs text-right">Acciones</TableHead>
   385→                </TableRow>
   386→              </TableHeader>
   387→              <TableBody>
   388→                {isLoading && Array.from({ length: 8 }).map((_, i) => (
   389→                  <TableRow key={i}><TableCell colSpan={7}><Skeleton className="h-8 w-full" /></TableCell></TableRow>
   390→                ))}
   391→                {!isLoading && sortedInventory.length === 0 && (
   392→                  <TableRow><TableCell colSpan={7} className="text-center text-muted-foreground py-8">Sin productos en inventario</TableCell></TableRow>
   393→                )}
   394→                {!isLoading && sortedInventory.map((p, idx) => {
   395→                  const margen = p.precioVenta > 0 ? ((p.precioVenta - p.costoUnitario) / p.precioVenta * 100) : 0
   396→                  return (
   397→                    <TableRow key={p.id} className={cn('hover:bg-muted/50 cursor-pointer', idx % 2 === 1 && 'bg-muted/20')} onClick={() => openSheet(p)}>
   398→                      <TableCell className="text-xs font-medium py-2 max-w-[200px] truncate">{p.nombre}</TableCell>
   399→                      <TableCell className="text-xs font-mono py-2">{p.sku}</TableCell>
   400→                      <TableCell className="text-xs text-center font-mono py-2">{p.totalStock}</TableCell>
   401→                      <TableCell className="text-xs text-right py-2">{formatCurrency(p.valorTotal)}</TableCell>
   402→                      <TableCell className="text-xs text-right py-2">
   403→                        <span className={cn(
   404→                          'font-mono',
   405→                          margen >= 30 ? 'text-emerald-600 dark:text-emerald-400' :
   406→                          margen >= 15 ? 'text-amber-600 dark:text-amber-400' :
   407→                          'text-red-600 dark:text-red-400'
   408→                        )}>
   409→                          {margen.toFixed(1)}%
   410→                        </span>
   411→                      </TableCell>
   412→                      <TableCell className="text-xs text-center py-2"><StatusBadge status={p.status} /></TableCell>
   413→                      <TableCell className="text-xs text-right py-2">
   414→                        <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   415→                          <Button size="icon" variant="ghost" className="h-7 w-7" title="Ver detalle" onClick={() => openSheet(p)}>
   416→                            <Eye className="h-3.5 w-3.5" />
   417→                          </Button>
   418→                          <Button size="icon" variant="ghost" className="h-7 w-7" title="Ver ubicaciones" onClick={() => setSelectedProduct(p)}>
   419→                            <MapPin className="h-3.5 w-3.5" />
   420→                          </Button>
   421→                          <Button size="icon" variant="ghost" className="h-7 w-7" title="Ajustar stock" onClick={() => openAjuste(p)}>
   422→                            <SlidersHorizontal className="h-3.5 w-3.5" />
   423→                          </Button>
   424→                          {p.stocks.length > 1 && (
   425→                            <Button size="icon" variant="ghost" className="h-7 w-7" title="Trasladar" onClick={() => openTraslado(p)}>
   426→                              <ArrowRightLeft className="h-3.5 w-3.5" />
   427→                            </Button>
   428→                          )}
   429→                        </div>
   430→                      </TableCell>
   431→                    </TableRow>
   432→                  )
   433→                })}
   434→              </TableBody>
   435→            </Table>
   436→          </div>
   437→        </CardContent>
   438→      </Card>
   439→
   440→      {/* Stock by location dialog */}
   441→      <Dialog open={!!selectedProduct} onOpenChange={(open) => { if (!open) setSelectedProduct(null) }}>
   442→        <DialogContent className="max-w-md">
   443→          <DialogHeader className="dialog-header-accent"><DialogTitle>Stock por Ubicación</DialogTitle><DialogDescription>Detalle de stock por ubicación del producto</DialogDescription></DialogHeader>
   444→          {selectedProduct && (
   445→            <div className="space-y-3">
   446→              <p className="text-sm"><span className="text-muted-foreground">Producto:</span> <span className="font-medium">{selectedProduct.nombre}</span></p>
   447→              <p className="text-sm"><span className="text-muted-foreground">SKU:</span> <span className="font-mono">{selectedProduct.sku}</span></p>
   448→              <Table>
   449→                <TableHeader>
   450→                  <TableRow>
   451→                    <TableHead className="text-xs">Pasillo</TableHead>
   452→                    <TableHead className="text-xs">Estante</TableHead>
   453→                    <TableHead className="text-xs">Nivel</TableHead>
   454→                    <TableHead className="text-xs text-right">Cantidad</TableHead>
   455→                  </TableRow>
   456→                </TableHeader>
   457→                <TableBody>
   458→                  {(() => {
   459→                    const stocks = stockEntries.filter((s) => s.idProducto === selectedProduct.id)
   460→                    if (stocks.length === 0) return <TableRow><TableCell colSpan={4} className="text-center text-muted-foreground py-4">Sin stock</TableCell></TableRow>
   461→                    return stocks.map((s) => (
   462→                      <TableRow key={s.idUbicacion}>
   463→                        <TableCell className="text-xs py-2">{s.ubicacion.pasillo}</TableCell>
   464→                        <TableCell className="text-xs py-2">{s.ubicacion.estante}</TableCell>
   465→                        <TableCell className="text-xs py-2">{s.ubicacion.nivel}</TableCell>
   466→                        <TableCell className="text-xs text-right font-mono py-2">{s.cantidad}</TableCell>
   467→                      </TableRow>
   468→                    ))
   469→                  })()}
   470→                </TableBody>
   471→              </Table>
   472→            </div>
   473→          )}
   474→        </DialogContent>
   475→      </Dialog>
   476→
   477→      {/* Ajuste Dialog */}
   478→      <Dialog open={!!ajusteProduct} onOpenChange={(open) => { if (!open) setAjusteProduct(null) }}>
   479→        <DialogContent className="max-w-sm">
   480→          <DialogHeader className="dialog-header-accent"><DialogTitle>Ajustar Stock</DialogTitle><DialogDescription>Modificar la cantidad de stock del producto</DialogDescription></DialogHeader>
   481→          {ajusteProduct && (
   482→            <div className="space-y-4">
   483→              <div className="space-y-1">
   484→                <Label className="text-xs text-muted-foreground">Producto</Label>
   485→                <p className="text-sm font-medium">{ajusteProduct.nombre}</p>
   486→              </div>
