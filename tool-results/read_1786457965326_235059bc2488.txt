    37→interface StockEntry {
    38→  idProducto: number
    39→  idUbicacion: number
    40→  cantidad: number

    44→
    45→interface InventoryItem {
    46→  id: number
    47→  sku: string
    48→  nombre: string
    49→  descripcion: string | null
    50→  idCategoria: number | null
    51→  idMarca: number | null
    52→  costoUnitario: number

    54→  stockMinimo: number
    55→  totalStock: number
    56→  valorTotal: number
    57→  status: 'ok' | 'low' | 'out'

    60→  marca?: { id: number; nombre: string } | null

    63→function StatusBadge({ status }: { status: string }) {
    64→  if (status === 'out')
    65→    return (
    66→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-danger">
    67→        <XCircle className="h-3 w-3" /> Sin Stock
    68→      </span>
    69→    )
    70→  if (status === 'low')
    71→    return (
    72→      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-warning">
    73→        <AlertTriangle className="h-3 w-3" /> Bajo Stock
    74→      </span>
    75→    )
    76→  return (
    77→    <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold badge-success">
    78→      <Check className="h-3 w-3" /> OK
    79→    </span>
    80→  )
    81→}

    89→  // Batch selection state

    91→  const [batchDialogOpen, setBatchDialogOpen] = useState(false)
    92→  const [batchQty, setBatchQty] = useState('')
    93→  const [batchType, setBatchType] = useState<'add' | 'subtract'>('add')
    94→  const [batchLoading, setBatchLoading] = useState(false)
    95→
    96→  // Ajuste state

   100→  const [ajusteLoading, setAjusteLoading] = useState(false)
   101→

   118→  const { data: stockEntries = [] } = useQuery<StockEntry[]>({

   123→  function toggleSelectAll() {
   124→    if (selectedIds.size === sortedInventory.length) {
   125→      setSelectedIds(new Set())
   126→    } else {
   127→      setSelectedIds(new Set(sortedInventory.map((p) => p.id)))
   128→    }
   129→  }
   130→
   131→  function toggleSelect(id: number) {
   132→    const next = new Set(selectedIds)
   133→    if (next.has(id)) {
   134→      next.delete(id)
   135→    } else {
   136→      next.add(id)
   137→    }
   138→    setSelectedIds(next)
   139→  }
   140→
   141→  function clearSelection() {
   142→    setSelectedIds(new Set())
   143→  }

   145→  async function submitBatchAdjust() {
   146→    if (selectedIds.size === 0 || !batchQty) return
   147→    const qty = parseInt(batchQty, 10)
   148→    if (isNaN(qty) || qty <= 0) {
   149→      toast.error('Cantidad inválida')
   150→      return
   151→    }
   152→
   153→    // Get stock entries for selected products
   154→    const selectedItems = sortedInventory
   155→      .filter((p) => selectedIds.has(p.id))
   156→      .flatMap((p) =>
   157→        p.stocks.map((s) => ({ idProducto: p.id, idUbicacion: s.idUbicacion }))
   158→      )
   159→
   160→    if (selectedItems.length === 0) {
   161→      toast.error('No hay stock entries para los productos seleccionados')
   162→      return
   163→    }
   164→
   165→    setBatchLoading(true)
   166→    try {
   167→      const res = await fetch('/api/wms/stock/batch-adjust', {
   168→        method: 'POST',
   169→        headers: { 'Content-Type': 'application/json' },
   170→        body: JSON.stringify({ items: selectedItems, cantidad: qty, tipo: batchType }),
   171→      })
   172→      if (!res.ok) {
   173→        const err = await res.json()
   174→        throw new Error(err.error || 'Error al ajustar stock')
   175→      }
   176→      toast.success(`Stock ajustado para ${selectedIds.size} productos`)
   177→      setBatchDialogOpen(false)
   178→      setBatchQty('')
   179→      setSelectedIds(new Set())
   180→      invalidateInventory()
   181→    } catch (e: any) {
   182→      toast.error(e.message || 'Error al ajustar stock')
   183→    } finally {
   184→      setBatchLoading(false)
   185→    }
   186→  }

   190→    if (selected.length === 0) return
   191→    exportToCSV(
   192→      selected.map((p) => ({
   193→        producto: p.nombre,
   194→        sku: p.sku,
   195→        totalStock: p.totalStock,
   196→        valorTotal: p.valorTotal,
   197→      })),
   198→      'inventario-seleccion',
   199→      [
   200→        { key: 'producto', label: 'Producto' },
   201→        { key: 'sku', label: 'SKU' },
   202→        { key: 'totalStock', label: 'Stock Total' },
   203→        { key: 'valorTotal', label: 'Valor Total', format: 'currency' },
   204→      ],
   205→    )
   206→  }

   208→  const { data: allLocations = [] } = useQuery({
   209→    queryKey: ['all-locations'],

   211→  })

   213→  const { data: productMovements = [] } = useQuery({
   214→    queryKey: ['product-movements-sheet', sheetProduct?.id],

   225→    if (totalStock === 0) status = 'out'
   226→    else if (totalStock < p.stockMinimo) status = 'low'

   233→      totalProducts: d.totalProductos,
   234→      stockValue: d.valorTotalStock,
   235→      bajoStock: d.productosBajoStock?.count ?? 0,
   236→    })),
   237→  })
   238→
   239→  const totalValor = inventory.reduce((sum, p) => sum + p.valorTotal, 0)

   241→  function invalidateInventory() {
   242→    queryClient.invalidateQueries({ queryKey: ['products-inventory'] })
   243→    queryClient.invalidateQueries({ queryKey: ['all-stock-inv'] })
   244→    queryClient.invalidateQueries({ queryKey: ['inventory-stats'] })
   245→    queryClient.invalidateQueries({ queryKey: ['product-movements-sheet'] })
   246→  }

   248→  // --- Sorting ---
   249→  function handleSort(field: string) {
   250→    if (sortField === field) {
   251→      setSortDir(sortDir === 'asc' ? 'desc' : 'asc')
   252→    } else {
   253→      setSortField(field)
   254→      setSortDir('asc')
   255→    }
   256→  }
   257→  function getSortValue(item: InventoryItem, field: string): any {
   258→    if (field === 'margen') {
   259→      return item.precioVenta > 0 ? ((item.precioVenta - item.costoUnitario) / item.precioVenta * 100) : 0
   260→    }
   261→    return item[field as keyof InventoryItem]
   262→  }
   263→
   264→  const sortedInventory = [...inventory].sort((a, b) => {
   265→    let aVal: any = getSortValue(a, sortField)
   266→    let bVal: any = getSortValue(b, sortField)
   267→    if (typeof aVal === 'string') { aVal = aVal.toLowerCase(); bVal = (bVal as string).toLowerCase() }
   268→    if (aVal < bVal) return sortDir === 'asc' ? -1 : 1
   269→    if (aVal > bVal) return sortDir === 'asc' ? 1 : -1
   270→    return 0
   271→  })

   273→  // --- AJUSTE handlers ---
   274→  function openAjuste(item: InventoryItem) {
   275→    setAjusteProduct(item)
   276→    setAjusteStock(String(item.totalStock))
   277→    setAjusteMotivo('')

   280→  async function submitAjuste() {
   281→    if (!ajusteProduct || ajusteStock === '') return
   282→    const newStock = parseInt(ajusteStock, 10)
   283→    if (isNaN(newStock) || newStock < 0) {
   284→      toast.error('Cantidad inválida')
   285→      return
   286→    }
   287→
   288→    const locationId = ajusteProduct.stocks.length > 0
   289→      ? ajusteProduct.stocks[0].idUbicacion
   290→      : allLocations[0]?.id
   291→
   292→    if (!locationId) {
   293→      toast.error('No hay ubicaciones disponibles')
   294→      return
   295→    }
   296→
   297→    setAjusteLoading(true)
   298→    try {
   299→      const res = await fetch('/api/wms/movimientos', {
   300→        method: 'POST',
   301→        headers: { 'Content-Type': 'application/json' },
   302→        body: JSON.stringify({
   303→          idProducto: ajusteProduct.id,
   304→          idUbicacion: locationId,
   305→          idTipo: 3,
   306→          cantidad: newStock,
   307→          referencia: 'Ajuste manual',
   308→          observacion: ajusteMotivo || null,
   309→        }),
   310→      })
   311→      if (!res.ok) {
   312→        const err = await res.json()
   313→        throw new Error(err.error || 'Error al ajustar stock')
   314→      }
   315→      toast.success('Stock ajustado correctamente')
   316→      setAjusteProduct(null)
   317→      invalidateInventory()
   318→    } catch (e: any) {
   319→      toast.error(e.message || 'Error al ajustar stock')
   320→    } finally {
   321→      setAjusteLoading(false)
   322→    }
   323→  }
   324→
   325→  // --- TRASLADO handlers ---
   326→  function openTraslado(item: InventoryItem) {
   327→    if (item.stocks.length < 1) return
   328→    setTrasladoProduct(item)
   329→    setTrasladoOrigen(String(item.stocks[0].idUbicacion))
   330→    setTrasladoDestino('')
   331→    setTrasladoCantidad('')
   332→  }
   333→
   334→  function getOrigenStock(): number {
   335→    if (!trasladoProduct) return 0
   336→    const entry = trasladoProduct.stocks.find((s) => String(s.idUbicacion) === trasladoOrigen)
   337→    return entry?.cantidad ?? 0
   338→  }
   339→
   340→  async function submitTraslado() {
   341→    if (!trasladoProduct || !trasladoOrigen || !trasladoDestino || !trasladoCantidad) return
   342→    if (trasladoOrigen === trasladoDestino) {
   343→      toast.error('El origen y destino no pueden ser iguales')
   344→      return
   345→    }
   346→    const qty = parseInt(trasladoCantidad, 10)
   347→    if (isNaN(qty) || qty <= 0) {
   348→      toast.error('Cantidad inválida')
   349→      return
   350→    }
   351→    const origenStock = getOrigenStock()
   352→    if (qty > origenStock) {
   353→      toast.error(`Stock insuficiente en origen (disponible: ${origenStock})`)
   354→      return
   355→    }
   356→
   357→    setTrasladoLoading(true)
   358→    try {
   359→      const res = await fetch('/api/wms/stock/transfer', {
   360→        method: 'POST',
   361→        headers: { 'Content-Type': 'application/json' },
   362→        body: JSON.stringify({
   363→          idProducto: trasladoProduct.id,
   364→          idUbicacionOrigen: parseInt(trasladoOrigen, 10),
   365→          idUbicacionDestino: parseInt(trasladoDestino, 10),
   366→          cantidad: qty,
   367→        }),
   368→      })
   369→      if (!res.ok) {
   370→        const err = await res.json()
   371→        throw new Error(err.error || 'Error en traslado')
   372→      }
   373→      toast.success('Traslado realizado correctamente')
   374→      setTrasladoProduct(null)
   375→      invalidateInventory()
   376→    } catch (e: any) {
   377→      toast.error(e.message || 'Error en traslado')
   378→    } finally {
   379→      setTrasladoLoading(false)
   380→    }
   381→  }

   498→                {!isLoading && sortedInventory.map((p, idx) => {

   500→                  return (
